add_executable(ndis_oid ndis_oid.c)
target_include_directories(ndis_oid PRIVATE ${REACTOS_SOURCE_DIR}/drivers/network/ndis/include)
target_compile_definitions(ndis_oid PRIVATE NDIS_WRAPPER NDIS_MINIPORT_DRIVER)
if(MSVC)
    target_compile_options(ndis_oid PRIVATE /Gy /Gw)
else()
    target_compile_options(ndis_oid PRIVATE -ffunction-sections -fdata-sections)
endif()
set_module_type(ndis_oid win32cui)
add_importlibs(ndis_oid msvcrt kernel32)
