last_partition_resized=1
k1_stop_watchdogs=wdt dev watchdog@D4080000; wdt stop; wdt dev PMIC_WDT; wdt stop
k1_fdt=0x11000000
k1_eth1=/soc/ethernet@cac81000
k1_isa=rv64imafdcv_zicbom_zicboz_zicntr_zicond_zicsr_zifencei_zihintpause_zihpm_zfh_zfhmin_zba_zbb_zbc_zbs_zkt_zvfh_zvfhmin_zvkt_sscofpmf_sstc_svinval_svnapot_svpbmt
k1_fixup_cpus=for k1_cpu in 0 1 2 3 4 5 6 7; do fdt set /cpus/cpu@${k1_cpu} riscv,isa ${k1_isa}; fdt set /cpus/cpu@${k1_cpu} riscv,cbom-block-size <64>; fdt set /cpus/cpu@${k1_cpu} riscv,cboz-block-size <64>; done
k1_fixup_l2=fdt mknode /cpus l2-cache0 && fdt set /cpus/l2-cache0 compatible "cache" && fdt set /cpus/l2-cache0 cache-level <2> && fdt set /cpus/l2-cache0 cache-unified && fdt set /cpus/l2-cache0 cache-size <524288> && fdt set /cpus/l2-cache0 cache-block-size <64> && fdt set /cpus/l2-cache0 cache-sets <512> && fdt set /cpus/l2-cache0 phandle <0x7f00> && fdt mknode /cpus l2-cache1 && fdt set /cpus/l2-cache1 compatible "cache" && fdt set /cpus/l2-cache1 cache-level <2> && fdt set /cpus/l2-cache1 cache-unified && fdt set /cpus/l2-cache1 cache-size <524288> && fdt set /cpus/l2-cache1 cache-block-size <64> && fdt set /cpus/l2-cache1 cache-sets <512> && fdt set /cpus/l2-cache1 phandle <0x7f01>
k1_fixup_l1=for k1_cpu in 0 1 2 3 4 5 6 7; do fdt set /cpus/cpu@${k1_cpu} i-cache-size <32768>; fdt set /cpus/cpu@${k1_cpu} i-cache-block-size <64>; fdt set /cpus/cpu@${k1_cpu} i-cache-sets <128>; fdt set /cpus/cpu@${k1_cpu} d-cache-size <32768>; fdt set /cpus/cpu@${k1_cpu} d-cache-block-size <64>; fdt set /cpus/cpu@${k1_cpu} d-cache-sets <128>; done
k1_fixup_l2link=for k1_cpu in 0 1 2 3; do fdt set /cpus/cpu@${k1_cpu} next-level-cache <0x7f00>; done; for k1_cpu in 4 5 6 7; do fdt set /cpus/cpu@${k1_cpu} next-level-cache <0x7f01>; done
k1_fixup_soc=fdt set /soc dma-noncoherent && fdt set /soc dma-ranges <0x0 0x0 0x0 0x0 0x0 0x80000000>
k1_fixup_pmic=fdt set /soc/twsi8@d401d800/pmic@41 interrupts <64>
k1_fixup_thermal=fdt mknode /soc thermal@d4018000 && fdt set /soc/thermal@d4018000 compatible "spacemit,k1x-tsensor" && fdt set /soc/thermal@d4018000 reg <0x0 0xd4018000 0x0 0x100> && fdt set /soc/thermal@d4018000 interrupts <61>
k1_fixup_audio=fdt mknode /soc audio@d4026000 && fdt set /soc/audio@d4026000 compatible "spacemit,k1x-audio" && fdt set /soc/audio@d4026000 reg <0x0 0xd4026000 0x0 0x30 0x0 0xd4000000 0x0 0x4000 0x0 0xd4012000 0x0 0x38> && fdt get value k1_plic /soc/interrupt-controller@e0000000 phandle && fdt set /soc/audio@d4026000 interrupt-parent <${k1_plic}> && fdt set /soc/audio@d4026000 interrupts <72>
k1_fixup_wdt=fdt mknode /soc watchdog@d4080000 && fdt set /soc/watchdog@d4080000 compatible "spacemit,k1x-wdt" && fdt set /soc/watchdog@d4080000 reg <0x0 0xd4080000 0x0 0x100 0x0 0xd4051020 0x0 0x4>
k1_fixup_gpu=fdt get value k1_plic /soc/interrupt-controller@e0000000 phandle && fdt set /soc/dpu@c0340000 reg <0x0 0xc0340000 0x0 0x2a000 0x0 0xc0440000 0x0 0x2a000 0x0 0xcac00000 0x0 0x80000> && fdt set /soc/dpu@c0340000 reg-names "dsi" "hdmi" "gpu" && fdt set /soc/dpu@c0340000 interrupt-parent <${k1_plic}> && fdt set /soc/dpu@c0340000 interrupts <75 139 90> && fdt set /soc/dpu@c0340000 interrupt-names "gpu" "hdmi" "dsi"
k1_fixup_wlan=fdt set /soc/sdh@d4280800 status okay && fdt set /soc/sdh@d4280800 bus-width <4> && fdt set /soc/sdh@d4280800 non-removable && fdt set /soc/sdh@d4280800 clk-src-freq <204800000> && fdt mknode /soc/sdh@d4280800 rf-pwrseq && fdt set /soc/sdh@d4280800/rf-pwrseq compatible "spacemit,rf-pwrseq" && fdt set /soc/sdh@d4280800/rf-pwrseq pwr-gpios <0x1c 67 0> && fdt mknode /soc/sdh@d4280800/rf-pwrseq wlan-pwrseq && fdt set /soc/sdh@d4280800/rf-pwrseq/wlan-pwrseq compatible "spacemit,wlan-pwrseq" && fdt set /soc/sdh@d4280800/rf-pwrseq/wlan-pwrseq regon-gpios <0x1c 116 0>
k1_fixup_eth1=fdt mknode /soc ethernet@cac81000 && fdt set ${k1_eth1} compatible "spacemit,k1x-emac" && fdt set ${k1_eth1} reg <0x0 0xcac81000 0x0 0x420> && fdt set ${k1_eth1} ctrl-reg <0x3ec> && fdt set ${k1_eth1} dline-reg <0x3f0> && fdt set ${k1_eth1} phy-mode "rgmii" && fdt set ${k1_eth1} phy-addr <1> && fdt set ${k1_eth1} clk_tuning_enable && fdt set ${k1_eth1} clk-tuning-by-delayline && fdt set ${k1_eth1} tx-phase <90> && fdt set ${k1_eth1} rx-phase <73> && fdt set ${k1_eth1} ref-clock-from-phy && fdt set /aliases ethernet0 /soc/ethernet@cac80000 && fdt set /aliases ethernet1 ${k1_eth1}
k1_board_k1-x_deb1=run k1_fixup_wlan && run k1_fixup_eth1
k1_fixup_board=setenv k1_board; setexpr k1_board sub "^k1-?x?_" "k1-x_" "${product_name}"; if env exists k1_board_${k1_board}; then run k1_board_${k1_board}; else true; fi
k1_prepare_fdt=fdt addr ${fdtcontroladdr} && fdt move ${fdtcontroladdr} ${k1_fdt} 0x20000 && run k1_fixup_cpus && run k1_fixup_l2 && run k1_fixup_l1 && run k1_fixup_l2link && run k1_fixup_soc && run k1_fixup_pmic && run k1_fixup_thermal && run k1_fixup_wdt && run k1_fixup_audio && run k1_fixup_gpu && run k1_fixup_board
k1_boot_freeldr=load ${bootfs_devname} ${boot_devnum}:${bootfs_part} 0x10000000 EFI/BOOT/BOOTRISCV64.EFI && if run k1_prepare_fdt; then bootefi 0x10000000 ${k1_fdt}; else bootefi 0x10000000 ${fdtcontroladdr}; fi
bootcmd=run k1_stop_watchdogs; run k1_boot_freeldr
